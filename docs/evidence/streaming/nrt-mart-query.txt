--- mart_trade_ohlc_1m_nrt — the newest BTCUSDT bars --------------------------
SELECT window_start, close, round(return_pct, 4) AS return_pct,
       round(gap_bps, 2) AS gap_bps, round(range_bps, 2) AS range_bps,
       round(sell_share, 3) AS sell_share, follows_previous,
       round(vwap_sma_5, 2) AS vwap_sma_5, round(vol_5m, 4) AS vol_5m
FROM mart_trade_ohlc_1m_nrt WHERE symbol = 'BTCUSDT'
ORDER BY window_start DESC LIMIT 8
